  baseURL: "https://interlock.example/v1",
  apiKey: process.env.INTERLOCK_KEY,
});
const res = await client.chat.completions
  .create({
    model: "larkspur-support",
    messages: [{
      role: "user",
      content: "Can I get a refund for last month?",

  });
